cmake_minimum_required(VERSION 3.24)
project(TVTDataBroadcastingRemote VERSION 0.4.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network WebSockets)
qt_standard_project_setup(REQUIRES 6.8)
option(REMOTE_BUILD_APP "Build the Qt Quick remote" ON)
option(REMOTE_BUILD_TESTS "Build integration tests" ON)
option(REMOTE_BUILD_ICON_TOOLS "Build SVG icon export utility" OFF)
if(REMOTE_BUILD_ICON_TOOLS AND NOT ANDROID)
    find_package(Qt6 REQUIRED COMPONENTS Gui Svg)
    qt_add_executable(remote-render-icons EXCLUDE_FROM_ALL tools/render-icons.cpp)
    set_target_properties(remote-render-icons PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(remote-render-icons PRIVATE Qt6::Gui Qt6::Svg)
endif()
if(WIN32)
    enable_language(RC)
endif()

add_library(remote_protocol INTERFACE)
target_include_directories(remote_protocol INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/shared)
target_link_libraries(remote_protocol INTERFACE Qt6::Core)

add_library(remote_client src/app/RemoteClient.cpp src/app/RemoteClient.h)
target_include_directories(remote_client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/app)
target_link_libraries(remote_client PUBLIC remote_protocol Qt6::Network Qt6::WebSockets)

if(NOT ANDROID)
    add_library(remote_broker src/server/Broker.cpp src/server/Broker.h src/server/ServerControl.cpp src/server/ServerControl.h
        src/server/StartupSettings.cpp src/server/StartupSettings.h src/server/ServerSettings.cpp src/server/ServerSettings.h)
    target_include_directories(remote_broker PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/server)
    target_link_libraries(remote_broker PUBLIC remote_protocol Qt6::Network Qt6::WebSockets)
    qt_add_executable(tvt-remote-server src/server/main.cpp)
    target_link_libraries(tvt-remote-server PRIVATE remote_broker)
    target_compile_definitions(tvt-remote-server PRIVATE REMOTE_VERSION="${PROJECT_VERSION}")
    if(WIN32)
        find_package(Qt6 REQUIRED COMPONENTS Widgets)
        add_library(remote_server_ui src/server/SettingsWindow.cpp src/server/SettingsWindow.h)
        target_include_directories(remote_server_ui PUBLIC src/server)
        target_link_libraries(remote_server_ui PUBLIC Qt6::Widgets)
        target_link_libraries(tvt-remote-server PRIVATE Qt6::Widgets)
        target_link_libraries(tvt-remote-server PRIVATE remote_server_ui)
        set_target_properties(tvt-remote-server PROPERTIES WIN32_EXECUTABLE TRUE)
        qt_add_resources(tvt-remote-server "server_icon" PREFIX "/icons" BASE "assets/icons" FILES assets/icons/server.png)
        target_sources(tvt-remote-server PRIVATE assets/icons/server.rc)
        target_include_directories(tvt-remote-server PRIVATE assets/icons)
        set_property(SOURCE assets/icons/server.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/server.ico")
    endif()
    qt_add_executable(tvt-remote-demo examples/qt-agent/main.cpp)
    set_target_properties(tvt-remote-demo PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(tvt-remote-demo PRIVATE remote_protocol Qt6::Network)
    if(WIN32)
        add_executable(tvt-remote-native-demo examples/native-agent/main.cpp)
        set_target_properties(tvt-remote-native-demo PROPERTIES AUTOMOC OFF AUTOUIC OFF)
        target_include_directories(tvt-remote-native-demo PRIVATE sdk third_party)
    endif()
    if(REMOTE_BUILD_TESTS)
        enable_testing()
        find_package(Qt6 REQUIRED COMPONENTS Test)
        qt_add_executable(remote-tests tests/integration.cpp)
        set_target_properties(remote-tests PROPERTIES WIN32_EXECUTABLE FALSE)
        target_link_libraries(remote-tests PRIVATE remote_broker remote_client Qt6::Test)
        add_dependencies(remote-tests tvt-remote-server)
        target_compile_definitions(remote-tests PRIVATE REMOTE_SERVER_PATH="$<TARGET_FILE:tvt-remote-server>")
        if(WIN32)
            target_include_directories(remote-tests PRIVATE sdk third_party)
            target_link_libraries(remote-tests PRIVATE ole32 shell32)
            qt_add_executable(server-window-tests tests/server-window.cpp)
            set_target_properties(server-window-tests PROPERTIES WIN32_EXECUTABLE FALSE)
            target_link_libraries(server-window-tests PRIVATE remote_server_ui Qt6::Test)
            add_test(NAME server-window-visibility COMMAND server-window-tests -o window-test-results.txt,txt -o -,txt)
            set_tests_properties(server-window-visibility PROPERTIES TIMEOUT 15)
        endif()
        add_test(NAME remote-integration COMMAND remote-tests -o test-results.txt,txt -o -,txt)
        set_tests_properties(remote-integration PROPERTIES TIMEOUT 45)
    endif()
endif()

if(REMOTE_BUILD_APP)
    find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2)
    qt_add_executable(tvt-remote src/app/main.cpp)
    set_source_files_properties(src/app/Main.qml PROPERTIES QT_RESOURCE_ALIAS Main.qml)
    qt_add_qml_module(tvt-remote URI TVTRemote VERSION 1.0 QML_FILES src/app/Main.qml)
    target_link_libraries(tvt-remote PRIVATE remote_client Qt6::Quick Qt6::QuickControls2)
    qt_add_resources(tvt-remote "application_icon" PREFIX "/icons" BASE "assets/icons" FILES assets/icons/remote.png)
    if(WIN32)
        target_sources(tvt-remote PRIVATE assets/icons/remote.rc)
        target_include_directories(tvt-remote PRIVATE assets/icons)
        set_property(SOURCE assets/icons/remote.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/icons/remote.ico")
    endif()
    if(ANDROID)
        qt_add_resources(tvt-remote "japanese_font" PREFIX "/fonts" BASE "assets/fonts" FILES assets/fonts/NotoSansJP.ttf)
        set_target_properties(tvt-remote PROPERTIES
            QT_ANDROID_PACKAGE_NAME "io.github.nanamitm.tvtremote"
            QT_ANDROID_APP_NAME "TVTest Remote"
            QT_ANDROID_APP_ICON "@mipmap/ic_launcher"
            QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android"
            QT_ANDROID_VERSION_NAME ${PROJECT_VERSION}
            QT_ANDROID_VERSION_CODE 10
            QT_ANDROID_MIN_SDK_VERSION 28
            QT_ANDROID_TARGET_SDK_VERSION 35
            QT_ANDROID_COMPILE_SDK_VERSION 36
            QT_ANDROID_SDK_BUILD_TOOLS_REVISION 36.0.0)
        qt_add_android_permission(tvt-remote NAME android.permission.INTERNET)
    endif()
endif()
